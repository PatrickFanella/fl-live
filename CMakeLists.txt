cmake_minimum_required(VERSION 3.25)
project(fl_live VERSION 0.1.0 LANGUAGES CXX)

option(FL_LIVE_BUILD_RENDERER "Build the offline WAV example" ON)
option(FL_LIVE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(FL_LIVE_BUILD_PIPEWIRE "Build the Linux PipeWire backend" OFF)
option(FL_LIVE_PIPEWIRE_INTEGRATION_TEST "Run a private, hardware-free PipeWire graph" OFF)
option(FL_LIVE_BUILD_DESKTOP "Build the Qt Quick session application" OFF)
option(FL_LIVE_LIVE_QUANTUM_DIAGNOSTIC "Run the currently unqualified live-quantum continuity diagnostic" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
if(FL_LIVE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer preset requires GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

if(FL_LIVE_BUILD_PIPEWIRE)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3=1.6.9)
    add_library(fl_live_pipewire src/pipewire_backend.cpp)
    target_link_libraries(fl_live_pipewire PUBLIC fl_live_engine PRIVATE PkgConfig::PIPEWIRE)
    add_executable(fl-live-play apps/play.cpp)
    target_link_libraries(fl-live-play PRIVATE fl_live_pipewire)
endif()

add_library(fl_live_engine src/engine.cpp)
target_include_directories(fl_live_engine PUBLIC include)
target_compile_features(fl_live_engine PUBLIC cxx_std_20)
target_compile_options(fl_live_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion)

add_library(fl_live_document src/session_document.cpp)
target_include_directories(fl_live_document PUBLIC include)
target_compile_features(fl_live_document PUBLIC cxx_std_20)
target_compile_options(fl_live_document PRIVATE -Wall -Wextra -Wpedantic -Wconversion)

if(FL_LIVE_BUILD_RENDERER)
    find_package(PkgConfig REQUIRED)
    # Initial qualified baseline; upgrades are explicit changes.
    pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile=1.2.2)
    add_executable(fl-live-render apps/render.cpp)
    target_link_libraries(fl-live-render PRIVATE fl_live_engine PkgConfig::SNDFILE)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(engine-tests tests/engine_tests.cpp)
    target_link_libraries(engine-tests PRIVATE fl_live_engine)
    add_test(NAME engine-contracts COMMAND engine-tests)
    add_executable(document-tests tests/document_tests.cpp)
    target_link_libraries(document-tests PRIVATE fl_live_document)
    add_test(NAME session-document COMMAND document-tests)
    add_executable(controller-tests tests/controller_tests.cpp)
    target_link_libraries(controller-tests PRIVATE fl_live_engine Threads::Threads)
    add_test(NAME control-handoff COMMAND controller-tests)
    set_tests_properties(control-handoff PROPERTIES TIMEOUT 15)
    if(FL_LIVE_BUILD_RENDERER)
        find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
        add_test(NAME render-roundtrip COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/render_roundtrip.py $<TARGET_FILE:fl-live-render>)
    endif()
    if(FL_LIVE_BUILD_PIPEWIRE AND FL_LIVE_PIPEWIRE_INTEGRATION_TEST)
        find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
        add_executable(live-quantum-probe tests/live_quantum_probe.cpp)
        target_link_libraries(live-quantum-probe PRIVATE fl_live_pipewire)
        add_test(NAME pipewire-frame-coded-control COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipewire_integration.py $<TARGET_FILE:fl-live-play>
            --quantum 257 --live-probe $<TARGET_FILE:live-quantum-probe> --no-quantum-changes)
        set_tests_properties(pipewire-frame-coded-control PROPERTIES TIMEOUT 30)
        add_test(NAME pipewire-reference-control COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipewire_integration.py $<TARGET_FILE:fl-live-play>
            --quantum 257 --live-probe $<TARGET_FILE:live-quantum-probe>
            --reference-producer --no-quantum-changes)
        set_tests_properties(pipewire-reference-control PROPERTIES TIMEOUT 30)
        if(FL_LIVE_LIVE_QUANTUM_DIAGNOSTIC)
            add_test(NAME pipewire-live-quantums COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipewire_integration.py $<TARGET_FILE:fl-live-play>
                --quantum 257 --live-probe $<TARGET_FILE:live-quantum-probe>)
            set_tests_properties(pipewire-live-quantums PROPERTIES TIMEOUT 30)
            add_test(NAME pipewire-reference-live-quantums COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipewire_integration.py $<TARGET_FILE:fl-live-play>
                --quantum 257 --live-probe $<TARGET_FILE:live-quantum-probe> --reference-producer)
            set_tests_properties(pipewire-reference-live-quantums PROPERTIES TIMEOUT 30)
        endif()
        foreach(quantum IN ITEMS 257 1024 16384)
            add_test(NAME pipewire-quantum-${quantum} COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipewire_integration.py $<TARGET_FILE:fl-live-play>
                --quantum ${quantum})
            set_tests_properties(pipewire-quantum-${quantum} PROPERTIES TIMEOUT 30)
        endforeach()
        if(NOT FL_LIVE_BUILD_DESKTOP)
            add_test(NAME pipewire-private-graph COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipewire_integration.py $<TARGET_FILE:fl-live-play>)
            set_tests_properties(pipewire-private-graph PROPERTIES TIMEOUT 30)
        endif()
    endif()
endif()

if(FL_LIVE_BUILD_DESKTOP)
    if(NOT FL_LIVE_BUILD_PIPEWIRE)
        message(FATAL_ERROR "Desktop requires FL_LIVE_BUILD_PIPEWIRE")
    endif()
    find_package(Qt6 6.12 REQUIRED COMPONENTS Quick QuickControls2)
    set(CMAKE_AUTOMOC ON)
    add_library(fl_live_session apps/desktop/session.cpp apps/desktop/export.cpp apps/desktop/session.hpp apps/desktop/project.cpp apps/desktop/project.hpp)
    target_include_directories(fl_live_session PUBLIC apps/desktop)
    target_link_libraries(fl_live_session PUBLIC fl_live_pipewire Qt6::Quick)
    add_executable(fl-live apps/desktop/main.cpp)
    target_link_libraries(fl-live PRIVATE fl_live_session Qt6::QuickControls2)
    qt_add_resources(fl-live "ui" PREFIX "/ui" BASE apps/desktop FILES apps/desktop/Main.qml)
    if(BUILD_TESTING)
        find_package(Qt6 6.12 REQUIRED COMPONENTS Test)
        add_executable(desktop-backend-probe tests/desktop_backend_probe.cpp)
        target_link_libraries(desktop-backend-probe PRIVATE fl_live_session Qt6::QuickControls2 Qt6::Test)
        qt_add_resources(desktop-backend-probe "ui" PREFIX "/ui" BASE apps/desktop FILES apps/desktop/Main.qml)
        if(FL_LIVE_PIPEWIRE_INTEGRATION_TEST)
            # Extend the same owned graph instead of starting a second fixture.
            add_test(NAME pipewire-private-graph COMMAND ${Python3_EXECUTABLE}
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipewire_integration.py $<TARGET_FILE:fl-live-play> $<TARGET_FILE:desktop-backend-probe>)
            set_tests_properties(pipewire-private-graph PROPERTIES TIMEOUT 40)
        endif()
        add_executable(desktop-tests tests/desktop_tests.cpp)
        target_link_libraries(desktop-tests PRIVATE fl_live_session Qt6::QuickControls2 Qt6::Test)
        qt_add_resources(desktop-tests "ui" PREFIX "/ui" BASE apps/desktop FILES apps/desktop/Main.qml)
        add_test(NAME desktop-session COMMAND desktop-tests)
        set_tests_properties(desktop-session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 20)
    endif()
endif()
